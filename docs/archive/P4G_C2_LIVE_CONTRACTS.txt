   147	        error: &ArcAgi3TransportError,
   148	    ) -> Option<ArcAgi3TransportFailureDisposition> {
   149	        match error {
   150	            ArcAgi3TransportError::HttpTransport { disposition, .. }

   152	            | ArcAgi3TransportError::InvalidRemoteResponse { disposition, .. }
   153	            | ArcAgi3TransportError::SessionIdentityMismatch { disposition, .. }
   154	            | ArcAgi3TransportError::CognitiveInitializationRejected { disposition, .. }

   156	                Some(*disposition)

   159	            ArcAgi3TransportError::InvalidBaseUrl

   175	            Ok(())

   184	        }

   194	    fn complete_cognitive_step(
   195	        &mut self,
   196	        cognitive_step: ArcAgi3CognitiveInteractionStep,
   197	        confidence: CognitiveSignal,
   198	        next_completed_step_count: u64,
   199	    ) -> Result<ArcAgi3LiveCognitiveStep, ArcAgi3LiveEnvironmentError> {
   200	        let completion = ArcAgi3EnvironmentTransportBoundary::complete_pending(
   201	            &mut self.transport,
   202	            &mut self.cognitive_runtime,
   203	            cognitive_step.command(),
   204	            confidence,
   205	        );
   206	
   207	        let completion = match completion {
   208	            Ok(completion) => completion,
   209	
   210	            Err(error) => {
   211	                self.mark_transport_failure(&error);
   212	
   213	                return Err(ArcAgi3LiveEnvironmentError::Transport(error));
   214	            }
   215	        };
   216	
   217	        self.completed_cognitive_step_count = next_completed_step_count;
   218	
   219	        Ok(ArcAgi3LiveCognitiveStep {
   220	            cognitive_step,
   221	            completion,
   222	            completed_cognitive_step_count: next_completed_step_count,
   223	        })
   224	    }

   226	    pub fn execute_with<F>(
   227	        &mut self,
   228	        confidence: CognitiveSignal,
   229	        begin: F,
   230	    ) -> Result<ArcAgi3LiveCognitiveStep, ArcAgi3LiveEnvironmentError>
   231	    where
   232	        F: FnOnce(
   233	            &mut ArcAgi3CognitiveInteractionRuntime,
   234	        )
   235	            -> Result<ArcAgi3CognitiveInteractionStep, ArcAgi3CognitiveInteractionError>,
   236	    {
   237	        self.ensure_active()?;
   238	
   239	        let next_completed_step_count = self
   240	            .completed_cognitive_step_count
   241	            .checked_add(1)
   242	            .ok_or(ArcAgi3LiveEnvironmentError::CognitiveStepCounterOverflow)?;
   243	
   244	        let cognitive_step = begin(&mut self.cognitive_runtime)?;
   245	
   246	        self.complete_cognitive_step(cognitive_step, confidence, next_completed_step_count)
   247	    }

   282	        let completion = match completion {

   287	
   288	                return Err(ArcAgi3LiveEnvironmentError::Transport(error));
   289	            }
   290	        };
   291

    95	#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
    96	pub struct ArcAgi3BoundedEpisodeRuntime;
    97	
    98	impl ArcAgi3BoundedEpisodeRuntime {
    99	    fn terminal_result(
   100	        runtime_status: ArcAgi3LiveEnvironmentStatus,
   101	        starting_completed_cognitive_step_count: u64,
   102	        steps: Vec<ArcAgi3LiveCognitiveStep>,
   103	        ending_completed_cognitive_step_count: u64,
   104	    ) -> Option<ArcAgi3BoundedEpisodeResult> {
   105	        let termination = match runtime_status {
   106	            ArcAgi3LiveEnvironmentStatus::Won => ArcAgi3BoundedEpisodeTermination::Won,
   107	            ArcAgi3LiveEnvironmentStatus::GameOver => ArcAgi3BoundedEpisodeTermination::GameOver,
   108	            ArcAgi3LiveEnvironmentStatus::NotStarted
   109	            | ArcAgi3LiveEnvironmentStatus::Active
   110	            | ArcAgi3LiveEnvironmentStatus::FaultedPending => {
   111	                return None;
   112	            }
   113	        };
   114	
   115	        Some(ArcAgi3BoundedEpisodeResult {
   116	            termination,
   117	            steps,
   118	            starting_completed_cognitive_step_count,
   119	            ending_completed_cognitive_step_count,
   120	            final_status: runtime_status,
   121	        })
   122	    }
   123	
   124	    fn verify_episode_step_counter<T>(
   125	        runtime: &ArcAgi3LiveEnvironmentRuntime<T>,
   126	        starting_completed_cognitive_step_count: u64,
   127	        completed_steps_in_episode: usize,
   128	    ) -> Result<(), ArcAgi3BoundedEpisodeError>
   129	    where
   130	        T: ArcAgi3EnvironmentTransport,
   131	    {
   132	        let completed_steps_in_episode = u64::try_from(completed_steps_in_episode)
   133	            .map_err(|_| ArcAgi3BoundedEpisodeError::StepHistoryLengthOverflow)?;
   134	
   135	        let expected = starting_completed_cognitive_step_count
   136	            .checked_add(completed_steps_in_episode)
   137	            .ok_or(ArcAgi3BoundedEpisodeError::ExpectedStepCounterOverflow)?;
   138	
   139	        let actual = runtime.completed_cognitive_step_count();
   140	
   141	        if actual != expected {
   142	            return Err(ArcAgi3BoundedEpisodeError::StepCounterMismatch { expected, actual });
   143	        }
   144	
   145	        Ok(())
   146	    }
   147	
   148	    pub fn run_with<T, F>(
   149	        runtime: &mut ArcAgi3LiveEnvironmentRuntime<T>,
   150	        policy: ArcAgi3BoundedEpisodePolicy,
   151	        mut execute_step: F,
   152	    ) -> Result<ArcAgi3BoundedEpisodeResult, ArcAgi3BoundedEpisodeError>
   153	    where
   154	        T: ArcAgi3EnvironmentTransport,
   155	        F: FnMut(
   156	            &mut ArcAgi3LiveEnvironmentRuntime<T>,
   157	        ) -> Result<ArcAgi3LiveCognitiveStep, ArcAgi3LiveEnvironmentError>,
   158	    {
   159	        let starting_completed_cognitive_step_count = runtime.completed_cognitive_step_count();
   160	
   161	        match runtime.status() {
   162	            ArcAgi3LiveEnvironmentStatus::Won | ArcAgi3LiveEnvironmentStatus::GameOver => {
   163	                return Ok(Self::terminal_result(
   164	                    runtime.status(),
   165	                    starting_completed_cognitive_step_count,
   166	                    Vec::new(),
   167	                    runtime.completed_cognitive_step_count(),
   168	                )
   169	                .expect("terminal status has terminal result"));
   170	            }
   171	
   172	            ArcAgi3LiveEnvironmentStatus::NotStarted => {
   173	                return Err(ArcAgi3BoundedEpisodeError::RuntimeNotRunnable(
   174	                    ArcAgi3LiveEnvironmentStatus::NotStarted,
   175	                ));
   176	            }
   177	
   178	            ArcAgi3LiveEnvironmentStatus::FaultedPending => {
   179	                return Err(ArcAgi3BoundedEpisodeError::RuntimeFaultedPending(
   180	                    runtime.fault_disposition(),
   181	                ));
   182	            }
   183	
   184	            ArcAgi3LiveEnvironmentStatus::Active => {}
   185	        }
   186	
   187	        let mut steps = Vec::with_capacity(policy.max_cognitive_steps());
   188	
   189	        while steps.len() < policy.max_cognitive_steps() {
   190	            let status_before_step = runtime.status();
   191	
   192	            match status_before_step {
   193	                ArcAgi3LiveEnvironmentStatus::Won | ArcAgi3LiveEnvironmentStatus::GameOver => {
   194	                    return Ok(Self::terminal_result(
   195	                        status_before_step,
   196	                        starting_completed_cognitive_step_count,
   197	                        steps,
   198	                        runtime.completed_cognitive_step_count(),
   199	                    )
   200	                    .expect("terminal status has terminal result"));
   201	                }
   202	
   203	                ArcAgi3LiveEnvironmentStatus::FaultedPending => {
   204	                    return Err(ArcAgi3BoundedEpisodeError::RuntimeFaultedPending(
   205	                        runtime.fault_disposition(),
   206	                    ));
   207	                }
   208	
   209	                ArcAgi3LiveEnvironmentStatus::NotStarted => {
   210	                    return Err(ArcAgi3BoundedEpisodeError::RuntimeNotRunnable(
   211	                        ArcAgi3LiveEnvironmentStatus::NotStarted,
   212	                    ));
   213	                }
   214	
   215	                ArcAgi3LiveEnvironmentStatus::Active => {}
   216	            }
   217	
   218	            let completed_before_failure = steps.len();
   219	
   220	            let step =
   221	                execute_step(runtime).map_err(|error| ArcAgi3BoundedEpisodeError::StepFailed {
   222	                    completed_steps_in_episode: completed_before_failure,
   223	                    error,
   224	                })?;
   225	
   226	            steps.push(step);
   227	
   228	            Self::verify_episode_step_counter(
   229	                runtime,
   230	                starting_completed_cognitive_step_count,
   231	                steps.len(),
   232	            )?;
   233	
   234	            match runtime.status() {
   235	                ArcAgi3LiveEnvironmentStatus::Won | ArcAgi3LiveEnvironmentStatus::GameOver => {
   236	                    return Ok(Self::terminal_result(
   237	                        runtime.status(),
   238	                        starting_completed_cognitive_step_count,
   239	                        steps,
   240	                        runtime.completed_cognitive_step_count(),
   241	                    )
   242	                    .expect("terminal status has terminal result"));
   243	                }
   244	
   245	                ArcAgi3LiveEnvironmentStatus::Active => {}
   246	
   247	                ArcAgi3LiveEnvironmentStatus::NotStarted
   248	                | ArcAgi3LiveEnvironmentStatus::FaultedPending => {
   249	                    return Err(ArcAgi3BoundedEpisodeError::UnexpectedPostStepStatus(
   250	                        runtime.status(),
   251	                    ));
   252	                }
   253	            }
   254	        }
   255	
   256	        Ok(ArcAgi3BoundedEpisodeResult {
   257	            termination: ArcAgi3BoundedEpisodeTermination::StepBudgetExhausted,
   258	            steps,
   259	            starting_completed_cognitive_step_count,
   260	            ending_completed_cognitive_step_count: runtime.completed_cognitive_step_count(),
   261	            final_status: runtime.status(),
   262	        })
   263	    }
   264	}

   226	impl ArcAgi3RestProtocol {
   227	    fn action_id_from_numeric(value: u8) -> Result<ArcAgi3ActionId, ArcAgi3RestDecodeError> {
   228	        match value {
   229	            0 => Ok(ArcAgi3ActionId::Reset),
   230	            1 => Ok(ArcAgi3ActionId::Action1),
   231	            2 => Ok(ArcAgi3ActionId::Action2),
   232	            3 => Ok(ArcAgi3ActionId::Action3),
   233	            4 => Ok(ArcAgi3ActionId::Action4),
   234	            5 => Ok(ArcAgi3ActionId::Action5),
   235	            6 => Ok(ArcAgi3ActionId::Action6),
   236	            7 => Ok(ArcAgi3ActionId::Action7),
   237	            _ => Err(ArcAgi3RestDecodeError::InvalidActionId(value)),
   238	        }
   239	    }
   240	
   241	    fn decode_action(
   242	        action: ArcAgi3WireActionInput,
   243	    ) -> Result<ArcAgi3Action, ArcAgi3RestDecodeError> {
   244	        let id = Self::action_id_from_numeric(action.id)?;
   245	
   246	        let x = action.data.as_ref().and_then(|data| data.x);
   247	        let y = action.data.as_ref().and_then(|data| data.y);
   248	
   249	        match id {
   250	            ArcAgi3ActionId::Action6 => {
   251	                let (Some(x), Some(y)) = (x, y) else {
   252	                    return Err(ArcAgi3RestDecodeError::InvalidActionPayload);
   253	                };
   254	
   255	                ArcAgi3Action::coordinate(x, y).ok_or(ArcAgi3RestDecodeError::InvalidActionPayload)
   256	            }
   257	
   258	            ArcAgi3ActionId::Reset => {
   259	                if x.is_some() || y.is_some() {
   260	                    return Err(ArcAgi3RestDecodeError::InvalidActionPayload);
   261	                }
   262	
   263	                Ok(ArcAgi3Action::reset())
   264	            }
   265	
   266	            _ => {
   267	                if x.is_some() || y.is_some() {
   268	                    return Err(ArcAgi3RestDecodeError::InvalidActionPayload);
   269	                }
   270	
   271	                ArcAgi3Action::discrete(id).ok_or(ArcAgi3RestDecodeError::InvalidActionPayload)
   272	            }
   273	        }
   274	    }
   275	
   276	    fn decode_state(state: &str) -> Result<ArcAgi3GameState, ArcAgi3RestDecodeError> {
   277	        match state {
   278	            "NOT_STARTED" => Ok(ArcAgi3GameState::NotPlayed),
   279	            "NOT_FINISHED" => Ok(ArcAgi3GameState::NotFinished),
   280	            "WIN" => Ok(ArcAgi3GameState::Win),
   281	            "GAME_OVER" => Ok(ArcAgi3GameState::GameOver),
   282	            _ => Err(ArcAgi3RestDecodeError::InvalidState(state.to_string())),
   283	        }
   284	    }

   307	        let action = command.action();

   323	        } else {
   324	            None
   325	        };
   326

   336	    pub fn decode_observation_json(
   337	        json: &str,
   338	    ) -> Result<ArcAgi3RestObservation, ArcAgi3RestDecodeError> {
   339	        let wire: ArcAgi3WireObservation = serde_json::from_str(json)
   340	            .map_err(|error| ArcAgi3RestDecodeError::InvalidJson(error.to_string()))?;
   341	
   342	        let game_id =
   343	            ArcAgi3GameId::new(wire.game_id).ok_or(ArcAgi3RestDecodeError::InvalidGameId)?;
   344	
   345	        if wire.guid.is_empty() || wire.guid.chars().any(char::is_whitespace) {
   346	            return Err(ArcAgi3RestDecodeError::InvalidGuid);
   347	        }
   348	
   349	        let state = Self::decode_state(&wire.state)?;
   350	
   351	        let frames = wire
   352	            .frame
   353	            .into_iter()
   354	            .map(|rows| ArcAgi3Grid::from_rows(rows).ok_or(ArcAgi3RestDecodeError::InvalidFrame))
   355	            .collect::<Result<Vec<_>, _>>()?;
   356	
   357	        let frames =
   358	            ArcAgi3FrameSequence::new(frames).ok_or(ArcAgi3RestDecodeError::InvalidFrame)?;
   359	
   360	        let available_actions = wire
   361	            .available_actions
   362	            .into_iter()
   363	            .map(Self::action_id_from_numeric)
   364	            .collect::<Result<Vec<_>, _>>()?;
   365	
   366	        if available_actions.contains(&ArcAgi3ActionId::Reset) {
   367	            return Err(ArcAgi3RestDecodeError::InvalidAvailableActions);
   368	        }
   369	
   370	        let available_actions = ArcAgi3AvailableActions::new(available_actions)
   371	            .ok_or(ArcAgi3RestDecodeError::InvalidAvailableActions)?;
   372	
   373	        let last_action = wire.action_input.map(Self::decode_action).transpose()?;
   374	
   375	        let levels_completed = wire
   376	            .levels_completed
   377	            .try_into()
   378	            .map_err(|_| ArcAgi3RestDecodeError::InvalidFrame)?;
   379	
   380	        let win_levels = wire
   381	            .win_levels
   382	            .try_into()
   383	            .map_err(|_| ArcAgi3RestDecodeError::InvalidFrame)?;
   384	
   385	        Ok(ArcAgi3RestObservation {
   386	            observation: ArcAgi3Observation::new(
   387	                game_id,
   388	                state,
   389	                frames,
   390	                levels_completed,
   391	                win_levels,
   392	                available_actions,
   393	                last_action,
   394	            ),
   395	            guid: wire.guid,
   396	        })
   397	    }

   415	    base_url: String,
   416	    api_key: String,

   419	
   420	impl ArcAgi3RestTransport {
   421	    pub fn new(base_url: String, api_key: String) -> Result<Self, ArcAgi3TransportError> {
   422	        let parsed =
   423	            reqwest::Url::parse(&base_url).map_err(|_| ArcAgi3TransportError::InvalidBaseUrl)?;
   424	
   425	        if parsed.scheme() != "http" && parsed.scheme() != "https" {
   426	            return Err(ArcAgi3TransportError::InvalidBaseUrl);
   427	        }
   428	
   429	        if api_key.is_empty() || api_key.chars().any(char::is_whitespace) {
   430	            return Err(ArcAgi3TransportError::EmptyApiKey);
   431	        }
   432	
   433	        let client = Client::builder()
   434	            .cookie_store(true)
   435	            .build()
   436	            .map_err(|error| ArcAgi3TransportError::HttpTransport {
   437	                message: error.to_string(),
   438	                disposition: ArcAgi3TransportFailureDisposition::NotDispatched,
   439	            })?;
   440	
   441	        Ok(Self {
   442	            client,
   443	            base_url: base_url.trim_end_matches('/').to_string(),
   444	            api_key,
   445	            session: None,
   446	        })
   447	    }
   448	
   449	    pub fn session(&self) -> Option<&ArcAgi3RestSession> {
   450	        self.session.as_ref()
   451	    }
   452	
   453	    fn send(
   454	        &self,
   455	        request: &ArcAgi3RestRequest,
   456	    ) -> Result<ArcAgi3RestObservation, ArcAgi3TransportError> {
   457	        let response = self
   458	            .client
   459	            .post(format!("{}{}", self.base_url, request.endpoint()))
   460	            .header("X-API-Key", &self.api_key)
   461	            .json(&request.wire_body())
   462	            .send()
   463	            .map_err(|error| ArcAgi3TransportError::HttpTransport {
   464	                message: error.to_string(),
   465	                disposition: ArcAgi3TransportFailureDisposition::DispatchIndeterminate,
   466	            })?;
   467	
   468	        let status = response.status();
   469	
   470	        let body = response
   471	            .text()
   472	            .map_err(|error| ArcAgi3TransportError::HttpTransport {
   473	                message: error.to_string(),
   474	                disposition: ArcAgi3TransportFailureDisposition::DispatchIndeterminate,
   475	            })?;
   476	
   477	        if !status.is_success() {
   478	            let disposition = if status.is_client_error() {
   479	                ArcAgi3TransportFailureDisposition::RejectedByEnvironment
   480	            } else {
   481	                ArcAgi3TransportFailureDisposition::DispatchIndeterminate
   482	            };
   483	
   484	            return Err(ArcAgi3TransportError::HttpStatus {
   485	                status: status.as_u16(),
   486	                body,
   487	                disposition,
   488	            });
   489	        }
   490	
   491	        ArcAgi3RestProtocol::decode_observation_json(&body).map_err(|error| {
   492	            ArcAgi3TransportError::InvalidRemoteResponse {
   493	                error,
   494	                disposition: ArcAgi3TransportFailureDisposition::DispatchIndeterminate,
   495	            }
   496	        })
   497	    }
   498	}

   501	    fn start_game(

   503	        game_id: &ArcAgi3GameId,
   504	        card_id: &str,

   238	pub struct ArcAgi3Grid {
   239	    width: usize,
   240	    height: usize,
   241	    cells: Vec<u8>,
   242	}
   243	
   244	impl ArcAgi3Grid {
   245	    pub fn from_rows(rows: Vec<Vec<u8>>) -> Option<Self> {
   246	        let height = rows.len();
   247	
   248	        if height == 0 || height > ARC_AGI_3_MAX_GRID_DIMENSION {
   249	            return None;
   250	        }
   251	
   252	        let width = rows.first()?.len();
   253	
   254	        if width == 0 || width > ARC_AGI_3_MAX_GRID_DIMENSION {
   255	            return None;
   256	        }
   257	
   258	        if rows.iter().any(|row| row.len() != width) {
   259	            return None;
   260	        }
   261	
   262	        let mut cells = Vec::with_capacity(width.saturating_mul(height));
   263	
   264	        for row in rows {
   265	            for cell in row {
   266	                if cell > ARC_AGI_3_MAX_CELL_VALUE {
   267	                    return None;
   268	                }
   269	
   270	                cells.push(cell);
   271	            }
   272	        }
   273	
   274	        Some(Self {
   275	            width,
   276	            height,
   277	            cells,
   278	        })
   279	    }
   280	
   281	    pub fn from_flat(width: usize, height: usize, cells: Vec<u8>) -> Option<Self> {
   282	        if width == 0
   283	            || height == 0
   284	            || width > ARC_AGI_3_MAX_GRID_DIMENSION
   285	            || height > ARC_AGI_3_MAX_GRID_DIMENSION
   286	        {
   287	            return None;
   288	        }
   289	
   290	        if width.checked_mul(height)? != cells.len() {
   291	            return None;
   292	        }
   293	
   294	        if cells.iter().any(|cell| *cell > ARC_AGI_3_MAX_CELL_VALUE) {
   295	            return None;
   296	        }
   297	
   298	        Some(Self {
   299	            width,
   300	            height,
   301	            cells,
   302	        })
   303	    }
   304	
   305	    pub fn width(&self) -> usize {
   306	        self.width
   307	    }
   308	
   309	    pub fn height(&self) -> usize {
   310	        self.height
   311	    }
   312	
   313	    pub fn cells(&self) -> &[u8] {
   314	        &self.cells
   315	    }
   316	
   317	    pub fn cell(&self, x: usize, y: usize) -> Option<u8> {
   318	        if x >= self.width || y >= self.height {
   319	            return None;
   320	        }
   321	
   322	        let index = y.checked_mul(self.width)?.checked_add(x)?;
   323	
   324	        self.cells.get(index).copied()
   325	    }
   326	
   327	    pub fn row(&self, y: usize) -> Option<&[u8]> {
   328	        if y >= self.height {
   329	            return None;
   330	        }
   331	
   332	        let start = y.checked_mul(self.width)?;
   333	        let end = start.checked_add(self.width)?;
   334	
   335	        self.cells.get(start..end)
   336	    }
   337	}
